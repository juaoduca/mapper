
query_doc = "{", { query }, "}";

query = query_name, [ query_args ], field_list;

query_name = identifier;

query_args = "(", boolExpression, ")";

boolExpression = orExpression;

orExpression = andExpression, { "OR", andExpression};

andExpression = notExpression, {"AND", notExpression};

notExpression = ["NOT"], comparison;

comparison = lvalue , [comparator], rvalue;

lvalue = identifier | function_call;

rvalue = identifier | function_call | value;

field_list = "{", { field }, "}";

field = [ alias, ":" ], field_name, [ field_args ], [ function_chain ], [ nested_object ];

field_name = identifier;

field_args = "(", [ field_arg ], ")";

field_arg = [ comparator ], rvalue;

function_chain = ".", function_call, { ".", function_call };

function_call = function_name, [ args ];

function_name = identifier;

args = "(", [ arg_list ], ")";

arg_list = arg, { ",", arg };

arg = value | identifier | function_call;

nested_object = [ join_type ], [ level_spec ], field_list;

join_type = ".leftjoin" | ".rightjoin";

level_spec = ".level", "(", number, ")" | ".all";

alias = identifier;

comparator = "=" | "<>" | "<" | ">" | "<=" | ">=" | "IS" , [ "NOT" ] , "NULL" | "LIKE"

value = string_literal | number | boolean_literal | null_literal;

identifier = letter, { letter | digit | "_" };

string_literal = '"', { character }, '"';

number = digit, { digit }, [ ".", { digit } ];

boolean_literal = "true" | "false";

null_literal = "null";
